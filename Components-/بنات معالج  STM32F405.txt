// الدليل الشامل والمدمج لوظائف البنات والتايمرات (Master Pinout & Timers Reference)
// المتحكم: STM32F405RGT6 (64-Pin)
// الاستخدام: مرجع لتصميم PCB وبرمجة طائرة الدرون المخصصة

/* -----------------------------------------------------------------
   📡 1. واجهات الاتصال التسلسلي (UART / USART)
   (تستخدم لـ: GPS، مستقبلات SBUS/ELRS، الكاميرات، Telemetry)
----------------------------------------------------------------- */
// [USART1] (أعلى سرعة) -> TX: PA9 أو PB6  | RX: PA10 أو PB7
// [USART2]             -> TX: PA2        | RX: PA3
// [USART3]             -> TX: PB10 أو PC10| RX: PB11 أو PC11
// [UART4]              -> TX: PA0 أو PC10 | RX: PA1 أو PC11
// [UART5]              -> TX: PC12       | RX: PD2
// [USART6] (أعلى سرعة) -> TX: PC6        | RX: PC7

/* -----------------------------------------------------------------
   🚀 2. واجهات الاتصال السريع (SPI)
   (تستخدم لـ: IMU، Gyro، رقائق OSD، Blackbox Flash)
----------------------------------------------------------------- */
// [SPI1] (أعلى سرعة) -> SCK: PA5 أو PB3 | MISO: PA6 أو PB4  | MOSI: PA7 أو PB5
// [SPI2]             -> SCK: PB10 أو PB13| MISO: PB14 أو PC2 | MOSI: PB15 أو PC3
// [SPI3]             -> SCK: PB3 أو PC10 | MISO: PB4 أو PC11 | MOSI: PB5 أو PC12

/* -----------------------------------------------------------------
   🧭 3. واجهات الاتصال (I2C)
   (تستخدم لـ: Barometer، Compass، شاشات OLED)
----------------------------------------------------------------- */
// [I2C1] -> SCL: PB6 أو PB8 | SDA: PB7 أو PB9
// [I2C2] -> SCL: PB10       | SDA: PB11
// [I2C3] -> SCL: PA8        | SDA: PC9

/* -----------------------------------------------------------------
   🚗 4. واجهة ناقل التحكم (CAN Bus)
   (تستخدم لـ: المحركات الذكية، الأنظمة الصناعية)
----------------------------------------------------------------- */
// [CAN1] -> TX: PA12 أو PB9 | RX: PA11 أو PB8
// [CAN2] -> TX: PB13 أو PB6 | RX: PB12 أو PB5

/* -----------------------------------------------------------------
   🔋 5. المداخل والمخارج التماثلية (ADC & DAC)
   (ADC: لقراءة حساسات التيار، أو مراقبة جهد بطارية 11.1V وتفادي تفريغها)
   (DAC: لتوليد إشارات تماثلية حقيقية)
----------------------------------------------------------------- */
// [ADC1 / ADC2 / ADC3] (مداخل قراءة الجهد):
// القنوات 0 إلى 7   -> PA0 إلى PA7
// القنوات 8 و 9     -> PB0 و PB1
// القنوات 10 إلى 15 -> PC0 إلى PC5

// [DAC] (مخارج الجهد التماثلي):
// المخرج 1 (OUT1) -> PA4
// المخرج 2 (OUT2) -> PA5

/* -----------------------------------------------------------------
   ⚙️ 6. التايمرات وقنوات الـ PWM (Timers & Channels)
   (تستخدم لـ: التحكم بسرعة محركات DJI، السيرفو، النغمات، وتوقيت النظام)
----------------------------------------------------------------- */
// --- تايمرات متقدمة (Advanced Control - 16-bit) ---
// [TIM1] -> CH1: PA8  | CH2: PA9  | CH3: PA10 | CH4: PA11
// [TIM8] -> CH1: PC6  | CH2: PC7  | CH3: PC8  | CH4: PC9

// --- تايمرات للأغراض العامة (General Purpose) ---
// [TIM2] (32-bit) -> CH1: PA0/PA15| CH2: PA1/PB3 | CH3: PA2/PB10 | CH4: PA3/PB11
// [TIM3] (16-bit) -> CH1: PA6/PC6 | CH2: PA7/PC7 | CH3: PB0/PC8  | CH4: PB1/PC9  (مستخدم لمحركات الدرون الأربعة)
// [TIM4] (16-bit) -> CH1: PB6     | CH2: PB7     | CH3: PB8      | CH4: PB9
// [TIM5] (32-bit) -> CH1: PA0     | CH2: PA1     | CH3: PA2      | CH4: PA3

// --- تايمرات إضافية (قناتين أو قناة واحدة) ---
// [TIM9]  -> CH1: PA2  | CH2: PA3
// [TIM10] -> CH1: PB8
// [TIM11] -> CH1: PB9
// [TIM12] -> CH1: PB14 | CH2: PB15 (PB14 ممتاز لبرمجة نغمات الـ Buzzer)
// [TIM13] -> CH1: PA6
// [TIM14] -> CH1: PA7
